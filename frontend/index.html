Main.java
import com.sun.net.httpserver.*; import java.net.InetSocketAddress; public class Main { public static void main(String[] args) throws Exception { HttpServer server = HttpServer.create(new InetSocketAddress(8080), 0); server.createContext("/scan", new ScanHandler()); server.start(); System.out.println("DarkLens running on http://localhost:8080"); } }
ScanHandler.java
public class ScanHandler implements HttpHandler { public void handle(HttpExchange ex) throws IOException { // Add CORS headers ex.getResponseHeaders().add("Access-Control-Allow-Origin", "*"); ex.getResponseHeaders().add("Content-Type", "application/json"); // Read request body → extract "url" String body = new String(ex.getRequestBody().readAllBytes()); String url = JsonUtil.extract(body, "url"); // Fetch HTML from target URL String html = WebFetcher.fetch(url); // Run pattern engine List<Finding> findings = PatternEngine.scan(html); // Serialize findings to JSON and respond String json = JsonSerializer.toJson(findings); ex.sendResponseHeaders(200, json.length()); ex.getResponseBody().write(json.getBytes()); } }
PatternEngine.java
public class PatternEngine { static List<Rule> rules = RuleLoader.load("patterns.json"); public static List<Finding> scan(String html) { List<Finding> findings = new ArrayList<>(); String lower = html.toLowerCase(); for (Rule rule : rules) { for (String keyword : rule.keywords) { int idx = lower.indexOf(keyword); if (idx != -1) { // Extract surrounding 120 chars as snippet String snippet = html.substring( Math.max(0, idx-20), Math.min(html.length(), idx+100) ); findings.add(new Finding( rule.name, rule.category, rule.severity, rule.description, snippet )); break; // one finding per rule } } } return findings; } }
