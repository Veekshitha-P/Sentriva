import com.sun.net.httpserver.*; import java.net.InetSocketAddress; import java.util.concurrent.Executors; public class Main { public static void main(String[] args) throws Exception { HttpServer server = HttpServer.create(new InetSocketAddress(8080), 0); server.createContext("/inspect", new InspectHandler()); server.setExecutor(Executors.newFixedThreadPool(4)); server.start(); System.out.println("[SENTRIVA] Inspection engine online → http://localhost:8080"); } }
InspectHandler.java
public class InspectHandler implements HttpHandler { public void handle(HttpExchange ex) throws IOException { ex.getResponseHeaders().add("Access-Control-Allow-Origin", "*"); ex.getResponseHeaders().add("Content-Type", "application/json"); if("OPTIONS".equals(ex.getRequestMethod())){ ex.sendResponseHeaders(204,-1); return; } String body = new String(ex.getRequestBody().readAllBytes()); String url = JsonUtil.extract(body, "url"); // Fetch HTML + headers once, shared across all engines FetchResult page = WebFetcher.fetch(url); // Run all 5 engines in parallel Map<String,List<Finding>> results = Map.of( "dark", new DarkPatternEngine().inspect(page), "security", new SecurityEngine().inspect(page), "perf", new PerformanceEngine().inspect(page), "a11y", new AccessibilityEngine().inspect(page), "privacy", new PrivacyEngine().inspect(page) ); String json = JsonSerializer.toJson(results); ex.sendResponseHeaders(200, json.getBytes().length); ex.getResponseBody().write(json.getBytes()); ex.close(); } }
DarkPatternEngine.java
public class DarkPatternEngine implements InspectionEngine { private final List<Rule> rules = RuleLoader.load("rules.json", "dark_patterns"); public List<Finding> inspect(FetchResult page) { List<Finding> out = new ArrayList<>(); String lower = page.html.toLowerCase(); for(Rule r : rules) { for(String kw : r.keywords) { int i = lower.indexOf(kw); if(i != -1) { String snip = page.html.substring(Math.max(0,i-20), Math.min(page.html.length(),i+110)); out.add(new Finding(r.name, r.category, r.severity, r.description, snip)); break; } } } return out; } }
SecurityEngine.java
public class SecurityEngine implements InspectionEngine { private static final String[][] CHECKS = { {"Content-Security-Policy", "CRITICAL", "No CSP — XSS attacks can execute scripts in users' browsers."}, {"Strict-Transport-Security","CRITICAL", "No HSTS — browser may allow HTTP downgrade attacks."}, {"X-Frame-Options", "HIGH", "No X-Frame-Options — page vulnerable to clickjacking."}, {"X-Content-Type-Options", "HIGH", "No XCTO — MIME sniffing attacks possible."}, {"Referrer-Policy", "MEDIUM", "No Referrer-Policy — sensitive URL params may leak."} }; public List<Finding> inspect(FetchResult page) { return Arrays.stream(CHECKS) .filter(c -> !page.headers.containsKey(c[0].toLowerCase())) .map(c -> new Finding("Missing: "+c[0], "Security Headers", c[1], c[2], "")) .collect(Collectors.toList()); } }
